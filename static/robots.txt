User-agent: *
Disallow:

# The contact endpoint only answers POST, and a GET returns an
# error. No crawler gains anything by fetching it, and keeping it
# out of the crawl means the rate limiter is not spending its
# budget on bots that were never going to submit the form.
#
# This is politeness, not protection: robots.txt is advisory and
# ignored by exactly the bots worth worrying about. The real
# defences (honeypot, rate limit, CRLF strip, origin check) live
# in static/api/contact.php and do not depend on this line.
Disallow: /api/
